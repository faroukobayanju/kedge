# Kedge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a CLI supervisor that runs headless Claude Code to turn failing dependency PRs green. After every step it detects silent drift, rolls back to the last good commit, and retries.

**Architecture:** A Python supervisor loop calls `claude -p` one step at a time inside a fresh clone of the PR. It runs the tests, applies deterministic drift checks, and either commits (a new good state) or resets with `git reset --hard` (a rollback), then appends a recovery note to the next prompt. neatlogs traces every step, Entire checkpoints every agent commit, and `steps.jsonl` drives an HTML and Markdown report.

**Tech Stack:** Python 3.12, `uv`, stdlib (`subprocess`, `argparse`, `string.Template`, `fnmatch`, `hashlib`), `neatlogs`, the `gh` CLI, the `claude` CLI, the `entire` CLI, `pytest`.

**Spec:** `docs/specs/kedge-spec.md`

## Global Constraints

- All code is committed between **Oct 10 12:30 PM IST and Oct 12 11:55 PM IST** in a new public repo `kedge`. No code is written before kickoff.
- Budget is $0: the default cap is `--budget 0.50` per PR, with `--max-rollbacks 3` and `--max-steps 6`. `over_budget` is always enforced, including with `--no-guard`.
- Kedge only runs PRs listed in `targets.json`, because tests run on the host with no Docker.
- PRs are always draft PRs, pushed to the user's fork and never upstream.
- Runtime dependency is only `neatlogs`. Everything else is stdlib or a CLI.
- Exit codes: `0` green, `1` gave up, `2` setup error.

## Review Focus

1. **Test runner prints no parseable summary** (crash on import, unknown runner). Expected: counts are `-1`, `tests_regressed` is skipped, and the loop keeps going. The test lives in Task 1.
2. **Worker returns bad JSON or times out** (`claude -p` non-zero exit, empty stdout). Expected: the step costs $0, is logged with `action: "worker_error"`, and counts toward `max_steps`. The test lives in Task 3.
3. **Worker changes no files.** Expected: no commit and no drift. Tests are re-run and the loop continues. The test lives in Task 4.
4. **Baseline already green** (someone fixed the PR in the meantime). Expected: exit 0 immediately and no PR is opened. The test lives in Task 4.
5. **The chaos-deleted test file is also outside the plan.** Expected: the drift is reported as `tests_regressed`, not `out_of_plan` (check order). The test lives in Task 1.

```python
# kedge/checks.py
import re
from fnmatch import fnmatch

NOTES = {
    "tests_regressed": "Previous attempt reduced passing/total tests. Do not delete or skip tests.",
    "out_of_plan": "Only edit files matching: {allowed}. You edited: {files}.",
    "loop": "You repeated a failed attempt. Try a different approach.",
}

def parse_summary(output: str) -> tuple[int, int]:
    jest = re.search(r"Tests:\s+(?:(\d+) failed, )?(\d+) passed, (\d+) total", output)
    if jest:
        return int(jest[2]), int(jest[3])
    nums = {k: int(v) for v, k in re.findall(r"(\d+) (passed|failed|errors?)\b", output)}
    if "passed" not in nums and "failed" not in nums:
        return -1, -1
    passed = nums.get("passed", 0)
    return passed, passed + nums.get("failed", 0) + nums.get("error", 0) + nums.get("errors", 0)

def _allowed(f: str, globs: list[str]) -> bool:
    # fnmatch '*' spans '/', but "src/**/x" must also match "src/x"
    return any(fnmatch(f, g) or fnmatch(f, g.replace("**/", "")) for g in globs)

def drift(*, passed, total, best_passed, baseline_total, files, allowed,
          diff_hash, seen_hashes, cost_total, budget, guard) -> str | None:
    if cost_total > budget:
        return "over_budget"
    if not guard:
        return None
    if diff_hash in seen_hashes:
        return "loop"
    if total >= 0 and (passed < best_passed or total < baseline_total):
        return "tests_regressed"
    if any(not _allowed(f, allowed) for f in files):
        return "out_of_plan"
    return None
```

```python
# kedge/gitgh.py
import hashlib, json, subprocess
from kedge.checks import parse_summary

def sh(args, cwd=None, check=True) -> str:
    return subprocess.run(args, cwd=cwd, check=check, capture_output=True, text=True).stdout

def head(cwd) -> str:
    return sh(["git", "rev-parse", "HEAD"], cwd).strip()

def changed_files(cwd) -> list[str]:
    out = sh(["git", "status", "--porcelain", "--untracked-files=all"], cwd)
    return [line[3:].split(" -> ")[-1] for line in out.splitlines() if line]

def diff_hash(cwd) -> str:
    sh(["git", "add", "-A", "-N"], cwd)  # intent-to-add so new files appear in diff
    return hashlib.sha256(sh(["git", "diff", "HEAD"], cwd).encode()).hexdigest()[:12]

def commit(cwd, msg) -> tuple[str, str]:
    sh(["git", "add", "-A"], cwd)
    sh(["git", "commit", "-qm", msg], cwd)
    cp = sh(["git", "log", "-1", "--format=%(trailers:key=Entire-Checkpoint,valueonly)"], cwd).strip()
    return head(cwd), cp

def reset(cwd, sha) -> None:
    sh(["git", "reset", "-q", "--hard", sha], cwd)
    sh(["git", "clean", "-qfd"], cwd)

def run_tests(cwd, test_cmd) -> tuple[bool, int, int]:
    p = subprocess.run(test_cmd, cwd=cwd, shell=True, capture_output=True, text=True, timeout=600)
    passed, total = parse_summary(p.stdout + p.stderr)
    return p.returncode == 0, passed, total

def clone_pr(pr_url, dest) -> dict:
    info = json.loads(sh(["gh", "pr", "view", pr_url, "--json", "title,baseRefName,headRefName,url"]))
    repo = "/".join(pr_url.split("/")[3:5])
    sh(["gh", "repo", "clone", repo, dest, "--", "-q"])
    sh(["gh", "pr", "checkout", pr_url.rsplit("/", 1)[1]], dest)
    return {"title": info["title"], "base": info["baseRefName"], "head": info["headRefName"], "repo": repo}

def ci_log(pr_url, head_branch, repo) -> str:
    runs = json.loads(sh(["gh", "run", "list", "-R", repo, "--branch", head_branch,
                          "--status", "failure", "--limit", "1", "--json", "databaseId"]))
    if not runs:
        return "(no failed CI run found)"
    log = sh(["gh", "run", "view", str(runs[0]["databaseId"]), "-R", repo, "--log-failed"], check=False)
    return "\n".join(log.splitlines()[-200:])

def open_draft_pr(cwd, fork, branch, title, body_path) -> str:
    sh(["git", "remote", "add", "fork", f"https://github.com/{fork}.git"], cwd, check=False)
    sh(["git", "push", "-qf", "fork", f"HEAD:{branch}"], cwd)
    base = json.loads(sh(["gh", "repo", "view", fork, "--json", "defaultBranchRef"]))["defaultBranchRef"]["name"]
    return sh(["gh", "pr", "create", "-R", fork, "--head", branch, "--base", base,
               "--draft", "--title", f"[kedge] {title}", "--body-file", body_path]).strip()
```

```python
# kedge/worker.py
import json, subprocess

FALLBACK_ALLOWED = ["package.json", "package-lock.json", "yarn.lock", "pnpm-lock.yaml",
                    "requirements*.txt", "pyproject.toml", "poetry.lock", "uv.lock", "src/**/*"]

def _claude(args, cwd) -> dict:
    p = subprocess.run(["claude", "-p", *args, "--output-format", "json"],
                       cwd=cwd, capture_output=True, text=True, timeout=900)
    return json.loads(p.stdout)

def run_edit_step(prompt, cwd) -> dict:
    try:
        d = _claude([prompt, "--permission-mode", "acceptEdits", "--max-turns", "15"], cwd)
        return {"cost_usd": float(d.get("total_cost_usd", 0.0)), "text": d.get("result", ""), "error": None}
    except (subprocess.TimeoutExpired, json.JSONDecodeError, OSError) as e:
        return {"cost_usd": 0.0, "text": "", "error": type(e).__name__}

def make_plan(title, ci_log, files, cwd) -> dict:
    prompt = (
        "A dependency-bump PR fails CI. Reply with ONLY a JSON object "
        '{"hypothesis": str, "allowed_files": [glob, ...]} listing the minimal file globs a fix may touch.\n'
        f"PR title: {title}\nFiles in PR: {files}\nFailing CI log tail:\n{ci_log[-4000:]}"
    )
    try:
        d = _claude([prompt, "--model", "haiku", "--max-turns", "1"], cwd)
    except (subprocess.TimeoutExpired, json.JSONDecodeError, OSError):
        d = {"result": "", "total_cost_usd": 0.0}
    cost = float(d.get("total_cost_usd", 0.0))
    text = d.get("result", "")
    try:
        plan = json.loads(text[text.index("{"): text.rindex("}") + 1])
        return {"hypothesis": plan["hypothesis"], "allowed_files": plan["allowed_files"], "cost_usd": cost}
    except (ValueError, KeyError):
        return {"hypothesis": "unknown", "allowed_files": FALLBACK_ALLOWED, "cost_usd": cost}
```

```python
PLAN = {"hypothesis": "fix import", "allowed_files": ["calc.py"], "cost_usd": 0.0}

def fixture(tmp_path, broken=True):
    subprocess.run(["git", "init", "-q", str(tmp_path)], check=True)
    for k, v in [("user.email", "k@k"), ("user.name", "k")]:
        subprocess.run(["git", "-C", str(tmp_path), "config", k, v], check=True)
    (tmp_path / "calc.py").write_text("from mathx import add\n" if broken else "def add(a, b): return a + b\n")
```

```python
# kedge/supervisor.py
from kedge import checks, gitgh, chaos as chaos_mod

def loop(cwd, test_cmd, plan, prompt_base, *, worker, guard=True, chaos=False, budget=0.5,
         max_rollbacks=3, max_steps=6, log=lambda rec: None, run_id="test") -> dict:
    ok, best, baseline_total = gitgh.run_tests(cwd, test_cmd)
    steps, notes, seen = [], [], set()
    cost, rollbacks, good = plan.get("cost_usd", 0.0), 0, gitgh.head(cwd)
    if ok:
        return {"status": "already_green", "steps": [], "cost_usd_total": cost, "rollbacks": 0}
    for step in range(1, max_steps + 1):
        prompt = prompt_base + "".join(f"\nRECOVERY NOTE: {n}" for n in notes)
        r = worker(prompt, cwd)
        cost += r["cost_usd"]
        if chaos and step == 1:
            chaos_mod.delete_one_test(cwd)
        ok, passed, total = gitgh.run_tests(cwd, test_cmd)
        files, h = gitgh.changed_files(cwd), gitgh.diff_hash(cwd)
        d = checks.drift(passed=passed, total=total, best_passed=best, baseline_total=baseline_total,
                         files=files, allowed=plan["allowed_files"], diff_hash=h, seen_hashes=seen,
                         cost_total=cost, budget=budget, guard=guard)
        rec = {"run_id": run_id, "step": step, "action": "worker_error" if r["error"] else "worker",
               "files_changed": files, "tests_passed": passed, "tests_total": total,
               "cost_usd": r["cost_usd"], "cost_usd_total": round(cost, 4), "drift": d,
               "rolled_back": False, "checkpoint_sha": good, "entire_checkpoint": ""}
        if d:
            gitgh.reset(cwd, good)
            seen.add(h)
            rec["rolled_back"] = d != "over_budget"
            steps.append(rec); log(rec)
            if d == "over_budget":
                break
            rollbacks += 1
            notes.append(checks.NOTES[d].format(allowed=plan["allowed_files"], files=files))
            if rollbacks > max_rollbacks:
                break
            continue
        if files:
            good, rec["entire_checkpoint"] = gitgh.commit(cwd, f"kedge {run_id} step {step}")
            rec["checkpoint_sha"] = good
            best = max(best, passed)
        steps.append(rec); log(rec)
        if ok:
            return {"status": "green", "steps": steps, "cost_usd_total": cost, "rollbacks": rollbacks}
    return {"status": "gave_up", "steps": steps, "cost_usd_total": cost, "rollbacks": rollbacks}
```

```python
# kedge/trace.py
import json, os, subprocess
from contextlib import contextmanager, nullcontext

_on = False

def start(run_id, tags) -> None:
    global _on
    if not os.environ.get("NEATLOGS_API_KEY"):
        return
    import neatlogs
    neatlogs.init(api_key=os.environ["NEATLOGS_API_KEY"], endpoint="https://ingest.neatlogs.com", workflow_name="dependency-pr-repair",
                  tags=[f"{k}:{v}" for k, v in {"run_id": run_id, **tags}.items()])
    _on = True

@contextmanager
def run_span(run_id, pr_url):
    """Root WORKFLOW span around one whole run, so the step spans nest under it."""
    if not _on:
        yield None; return
    import neatlogs
    with neatlogs.trace(f"kedge run {run_id}", kind="WORKFLOW") as s:
        s.set_attribute("kedge.run_id", run_id); s.set_attribute("kedge.pr_url", pr_url)
        yield s

def record(rec) -> None:
    if not _on:
        return
    import neatlogs
    kind = "AGENT" if rec["action"] in ("replan",) else "TOOL"
    with neatlogs.trace(f"step {rec['step']} {rec['action']}", kind=kind) as s:
        for k, v in rec.items():
            s.set_attribute(f"kedge.{k}", json.dumps(v) if isinstance(v, (list, dict)) or v is None else v)

def finish() -> None:
    if _on:
        import neatlogs
        neatlogs.flush(); neatlogs.shutdown()

def trace_url(run_id) -> str:
    # Readback uses the CLI with a service-account token (NEATLOGS_TOKEN), never the ingest key.
    try:
        env = {k: v for k, v in os.environ.items() if k != "NEATLOGS_API_KEY"}
        out = subprocess.run(["neatlogs", "traces", "list", "--workflow", "dependency-pr-repair", "--json"],
                             capture_output=True, text=True, timeout=20, env=env).stdout
        for t in json.loads(out).get("data", []):
            if f"run_id:{run_id}" in json.dumps(t):
                return t.get("url") or t["id"]
        return "unavailable"
    except Exception:
        return "unavailable"
```

```python
# kedge/report.py
import html, json, os
from string import Template

HTML = Template("""<!doctype html><meta charset="utf-8"><title>Kedge run $run_id</title>
<style>body{font:14px system-ui;margin:2rem;max-width:960px}td,th{padding:4px 8px;border-bottom:1px solid #ddd}
.drift{background:#fde8e8}</style>
<h1>Kedge run $run_id: $status</h1>
<p>PR: <a href="$pr_url">$pr_url</a> · cost $$$cost · rollbacks $rollbacks · <a href="$trace_url">neatlogs trace</a></p>
<table><tr><th>Step</th><th>Action</th><th>Files</th><th>Tests</th><th>Cost</th><th>Drift</th></tr>$rows</table>
<h2>Entire checkpoint</h2><pre>$entire</pre>""")

def _status(r):
    return f"{r['drift']} (rolled back)" if r.get("rolled_back") else (r.get("drift") or "ok")

def render(run_dir, meta) -> tuple[str, str]:
    recs = [json.loads(l) for l in open(os.path.join(run_dir, "steps.jsonl")) if l.strip()]
    rows = "".join(
        f'<tr class="{"drift" if r.get("drift") else ""}"><td>{r["step"]}</td><td>{r["action"]}</td>'
        f'<td>{html.escape(", ".join(r["files_changed"]))}</td><td>{r["tests_passed"]}/{r["tests_total"]}</td>'
        f'<td>${r["cost_usd"]:.3f}</td><td>{_status(r)}</td></tr>' for r in recs)
    page = HTML.substitute(run_id=meta["run_id"], status=meta["status"], pr_url=meta["pr_url"],
                           cost=f'{meta["cost_usd_total"]:.3f}', rollbacks=meta["rollbacks"],
                           trace_url=meta["trace_url"], rows=rows, entire=html.escape(meta["entire_explain"]))
    md = [f"## Kedge run `{meta['run_id']}`: {meta['status']}",
          f"Cost ${meta['cost_usd_total']:.3f} · rollbacks {meta['rollbacks']} · [neatlogs trace]({meta['trace_url']})", "",
          "| Step | Action | Files | Tests | Cost | Drift |", "|---|---|---|---|---|---|"]
    md += [f"| {r['step']} | {r['action']} | {', '.join(r['files_changed'])} | {r['tests_passed']}/{r['tests_total']} "
           f"| ${r['cost_usd']:.3f} | {_status(r).replace('(rolled back)', 'rolled back')} |" for r in recs]
    md += ["", "### Entire checkpoint", "```", meta["entire_explain"], "```"]
    hp, mp = os.path.join(run_dir, "report.html"), os.path.join(run_dir, "report.md")
    open(hp, "w").write(page); open(mp, "w").write("\n".join(md))
    return hp, mp
```

```python
# kedge/cli.py
import argparse, json, os, re, sys, tempfile, time, uuid
from kedge import gitgh, report, supervisor, trace, worker

PROMPT = ("You are fixing a failing dependency-bump PR.\nPR: {title}\nPlan: {hyp}\n"
          "Only edit files matching: {allowed}\nFailing CI log tail:\n{log}\n"
          "Do ONE step toward making the tests pass, then stop. Never delete or skip tests.")

def _targets(path="targets.json"):
    return {t["pr_url"]: t for t in json.load(open(path))} if os.path.exists(path) else {}

def run_one(pr_url, test_cmd, fork, *, guard, chaos, budget, max_rollbacks, max_steps) -> dict:
    run_id = time.strftime("%Y%m%d-%H%M") + "-" + uuid.uuid4().hex[:4]
    run_dir = os.path.abspath(os.path.join("runs", run_id)); os.makedirs(run_dir)
    cwd = os.path.join(tempfile.gettempdir(), "kedge", run_id)
    pr = gitgh.clone_pr(pr_url, cwd)
    gitgh.sh(["entire", "enable"], cwd, check=False)
    log_tail = gitgh.ci_log(pr_url, pr["head"], pr["repo"])
    plan = worker.make_plan(pr["title"], log_tail, gitgh.sh(["git", "diff", "--name-only", f"origin/{pr['base']}...HEAD"], cwd).split(), cwd)
    json.dump(plan, open(os.path.join(run_dir, "plan.json"), "w"))
    trace.start(run_id, {"pr_url": pr_url, "guard": "on" if guard else "off", "chaos": "on" if chaos else "off"})
    steps_f = open(os.path.join(run_dir, "steps.jsonl"), "a")
    def log(rec):
        steps_f.write(json.dumps(rec) + "\n"); steps_f.flush(); trace.record(rec)
    prompt = PROMPT.format(title=pr["title"], hyp=plan["hypothesis"], allowed=plan["allowed_files"], log=log_tail)
    res = supervisor.loop(cwd, test_cmd, plan, prompt, worker=worker.run_edit_step, guard=guard, chaos=chaos,
                          budget=budget, max_rollbacks=max_rollbacks, max_steps=max_steps, log=log, run_id=run_id)
    steps_f.close()
    cp = next((s["entire_checkpoint"] for s in reversed(res["steps"]) if s["entire_checkpoint"]), "")
    explain = gitgh.sh(["entire", "checkpoint", "explain", cp], cwd, check=False) if cp else "(no Entire checkpoint)"
    meta = {"run_id": run_id, "pr_url": pr_url, "status": res["status"], "cost_usd_total": res["cost_usd_total"],
            "rollbacks": res["rollbacks"], "trace_url": trace.trace_url(run_id), "entire_explain": explain}
    _, md = report.render(run_dir, meta)
    res.update(run_id=run_id, pr_url=pr_url, draft_pr=None)
    if res["status"] == "green":
        res["draft_pr"] = gitgh.open_draft_pr(cwd, fork, f"kedge/{run_id}", pr["title"], md)
    return res

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="kedge")
    sub = ap.add_subparsers(dest="cmd", required=True)
    for name in ("run", "batch"):
        p = sub.add_parser(name)
        p.add_argument("target")
        p.add_argument("--chaos", action="store_true")
        p.add_argument("--no-guard", action="store_true")
        p.add_argument("--budget", type=float, default=0.50)
        p.add_argument("--max-rollbacks", type=int, default=3)
        p.add_argument("--max-steps", type=int, default=6)
    a = ap.parse_args(argv)
    kw = dict(guard=not a.no_guard, chaos=a.chaos, budget=a.budget, max_rollbacks=a.max_rollbacks, max_steps=a.max_steps)
    if a.cmd == "run":
        if not re.fullmatch(r"https://github\.com/[^/]+/[^/]+/pull/\d+", a.target):
            print("error: expected https://github.com/<owner>/<repo>/pull/<n>", file=sys.stderr); return 2
        t = _targets().get(a.target)
        if not t:
            print("error: PR not in targets.json allowlist", file=sys.stderr); return 2
        res = run_one(a.target, t["test_cmd"], t["fork"], **kw)
        print(json.dumps({k: res[k] for k in ("run_id", "status", "cost_usd_total", "rollbacks", "draft_pr")}))
        return 0 if res["status"] in ("green", "already_green") else 1
    results = []
    for t in json.load(open(a.target)):
        try:
            r = run_one(t["pr_url"], t["test_cmd"], t["fork"], **kw)
        except Exception as e:  # one broken target must not stop the batch
            r = {"pr_url": t["pr_url"], "status": f"error: {e}", "cost_usd_total": 0.0, "rollbacks": 0, "steps": []}
        results.append(r)
        print(json.dumps({k: r.get(k) for k in ("pr_url", "status", "cost_usd_total", "rollbacks", "draft_pr")}))
    green = [r for r in results if r["status"] == "green"]
    drifts = sum(1 for r in results for s in r["steps"] if s["drift"] and s["drift"] != "over_budget")
    cost = sum(r["cost_usd_total"] for r in results)
    print(json.dumps({"green": len(green), "total": len(results), "drifts_caught": drifts,
                      "usd_per_green": round(cost / len(green), 3) if green else None}))
    return 0
```

```python
# kedge/supervisor.py
import json, os
from kedge import checks, gitgh, chaos as chaos_mod

def decide_next(history, max_rollbacks) -> str:
    if history and history[-1].get("drift") == "over_budget":
        return "give_up"
    if sum(1 for s in history if s.get("rolled_back")) > max_rollbacks:
        return "give_up"
    if len(history) >= 2 and all(s.get("rolled_back") for s in history[-2:]):
        return "replan"
    return "edit"

def lessons_for(path, repo, n=3) -> list[str]:
    if not os.path.exists(path):
        return []
    rows = [json.loads(l) for l in open(path) if l.strip()]
    return [r["note"] for r in rows][-n:]

def record_lessons(path, repo, steps, notes, status) -> None:
    # a note is a lesson only if it preceded a rollback that the run then recovered from
    if status != "green" or not any(s.get("rolled_back") for s in steps):
        return
    with open(path, "a") as f:
        for n in notes:
            f.write(json.dumps({"repo": repo, "note": n}) + "\n")

def _rec(run_id, step, action, s, **kw):
    rec = {"run_id": run_id, "step": step, "action": action, "files_changed": [], "tests_passed": -1,
           "tests_total": -1, "cost_usd": 0.0, "cost_usd_total": round(s["cost"], 4), "drift": None,
           "rolled_back": False, "checkpoint_sha": s["good"], "entire_checkpoint": "", "plan_step": s["plan_idx"]}
    rec.update(kw)
    return rec

def _finish(status, s):
    return {"status": status, "steps": s["steps"], "cost_usd_total": s["cost"],
            "rollbacks": sum(1 for x in s["steps"] if x.get("rolled_back")), "notes": s["notes"]}

def loop(cwd, test_cmd, plan, prompt_base, *, worker, replanner=None, guard=True, chaos=False, budget=0.5,
         max_rollbacks=3, max_steps=6, log=lambda rec: None, run_id="test", state=None, save=lambda s: None) -> dict:
    s = state
    if not s:
        ok, best, baseline_total = gitgh.run_tests(cwd, test_cmd)
        s = {"step": 0, "good": gitgh.head(cwd), "notes": [], "seen": [], "cost": plan.get("cost_usd", 0.0),
             "best": best, "baseline_total": baseline_total, "plan": plan, "plan_idx": 0, "steps": []}
        if ok:
            return _finish("already_green", s)
        save(s)
    else:
        gitgh.reset(cwd, s["good"])  # discard half-finished work from the crashed step
    while s["step"] < max_steps:
        action = decide_next(s["steps"], max_rollbacks)
        if action == "give_up":
            break
        if action == "replan" and replanner:
            s["plan"] = replanner(s["plan"], s["notes"], cwd)
            s["cost"] += s["plan"].get("cost_usd", 0.0)
            s["plan_idx"] = 0
            rec = _rec(run_id, s["step"], "replan", s)
            s["steps"].append(rec); log(rec); save(s)
            continue
        s["step"] += 1
        save(s)
        steps_list = s["plan"].get("steps") or []
        current = steps_list[min(s["plan_idx"], len(steps_list) - 1)] if steps_list else "make the tests pass"
        prompt = (prompt_base + f"\nCURRENT PLAN STEP: {current}"
                  + "".join(f"\nRECOVERY NOTE: {n}" for n in s["notes"][-3:]))
        r = worker(prompt, cwd)
        s["cost"] += r["cost_usd"]
        if chaos and s["step"] == 1:
            chaos_mod.delete_one_test(cwd)
        ok, passed, total = gitgh.run_tests(cwd, test_cmd)
        files, h = gitgh.changed_files(cwd), gitgh.diff_hash(cwd)
        d = checks.drift(passed=passed, total=total, best_passed=s["best"], baseline_total=s["baseline_total"],
                         files=files, allowed=s["plan"]["allowed_files"], diff_hash=h, seen_hashes=set(s["seen"]),
                         cost_total=s["cost"], budget=budget, guard=guard)
        if ok and not d and guard:  # verification: a green result must reproduce
            ok2, _, total2 = gitgh.run_tests(cwd, test_cmd)
            if not ok2 or (total2 >= 0 and total2 < s["baseline_total"]):
                d = "verify_failed"
        rec = _rec(run_id, s["step"], "worker_error" if r["error"] else "worker", s, files_changed=files,
                   tests_passed=passed, tests_total=total, cost_usd=r["cost_usd"], drift=d)
        if d:
            gitgh.reset(cwd, s["good"])
            s["seen"].append(h)
            rec["rolled_back"] = d != "over_budget"
            if d != "over_budget":
                s["notes"].append(checks.NOTES[d].format(allowed=s["plan"]["allowed_files"], files=files))
            s["steps"].append(rec); log(rec); save(s)
            continue
        if files:
            s["good"], rec["entire_checkpoint"] = gitgh.commit(cwd, f"kedge {run_id} step {s['step']}")
            rec["checkpoint_sha"] = s["good"]
            if passed > s["best"]:
                s["best"] = passed
                s["plan_idx"] += 1  # progress: move to the next plan step
        s["steps"].append(rec); log(rec); save(s)
        if ok:
            return _finish("green", s)
    return _finish("gave_up", s)
```

```python
def _plan_call(prompt, cwd) -> dict:
    try:
        d = _claude([prompt, "--model", "haiku", "--max-turns", "1"], cwd)
    except (subprocess.TimeoutExpired, json.JSONDecodeError, OSError):
        d = {"result": "", "total_cost_usd": 0.0}
    cost, text = float(d.get("total_cost_usd", 0.0)), d.get("result", "")
    try:
        p = json.loads(text[text.index("{"): text.rindex("}") + 1])
        return {"hypothesis": p["hypothesis"], "allowed_files": p["allowed_files"],
                "steps": p.get("steps") or ["make the failing tests pass"], "cost_usd": cost}
    except (ValueError, KeyError):

SCHEMA = '{"hypothesis": str, "allowed_files": [glob, ...], "steps": [1-4 short imperative steps]}'
```

```python
import re
ERR = re.compile(r"error|fail|traceback|exception|assert|cannot find|not found|undefined", re.I)

def trim_log(log, max_lines=60) -> str:
    lines = log.splitlines()
    keep = sorted({j for i, l in enumerate(lines) if ERR.search(l)
                   for j in range(max(0, i - 2), min(len(lines), i + 3))})
    picked = [lines[i] for i in keep] or lines
    return "\n".join(picked[-max_lines:])
```

```python
# kedge/shiplog.py
def header(branch, budget) -> str:
    return f"⚓ kedge  {branch}  ·  budget ${budget:.2f}"

def line(rec) -> str:
    if rec["action"] == "replan":
        return "🧭 replanning after 2 drifts"
    if rec.get("drift") == "over_budget":
        return "💸 out of budget, dropping anchor"
    tests = f"tests {rec['tests_passed']}/{rec['tests_total']}"
    if rec.get("drift"):
        files = ", ".join(rec["files_changed"][:3])
        return f"🌊 DRIFT  {rec['drift']} · {tests} · {files}\n⚓ hauled back to {rec['checkpoint_sha'][:7]}"
    if rec["action"] == "worker_error":
        return f"⛈  step {rec['step']}  worker error · {tests}"
    return f"⛵ step {rec['step']}  edited {len(rec['files_changed'])} files · {tests}"

def footer(res, trace_url) -> str:
    where = res.get("draft_pr") or "no PR"
    return f"🏁 {res['status']} · ${res['cost_usd_total']:.2f} · {where} · trace {trace_url}"
```

### Task 18: `false_claim`, catching the agent lying (A, Sun 10–10:30 PM)

```python
CLAIM = re.compile(r"all (?:the )?tests (?:now )?pass(?:ing)?|tests (?:are )?passing|\bfixed\b|\bresolved\b", re.I)
NEGATION = re.compile(r"\b(?:not|never|unable|cannot|can't|couldn't|still fail\w*)\b", re.I)

def claimed_success(text: str) -> str | None:
    # a negated sentence ("could not fix", "still failing") is not a claim
    if not text or NEGATION.search(text):

    m = CLAIM.search(text)
    return m.group(0) if m else None
```
